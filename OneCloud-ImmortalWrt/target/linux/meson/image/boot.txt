
setenv condev "console=ttyAML0,115200n8 no_console_suspend consoleblank=0"
setenv bootargs_emmc "root=/dev/mmcblk1p2 rootfstype=ext4 rootwait ro ${condev} fsck.repair=yes net.ifnames=0 biosdevname=0 panic=3 oops=panic mac=${mac}"
setenv kernel_loadaddr "0x00208000"
setenv dtb_loadaddr "0x21800000"
setenv initrd_loadaddr "0x22000000"
setenv dtb_name "dtb"

# Always set bootargs (not dependent on uInitrd presence)
setenv bootargs ${bootargs_emmc}

# uInitrd is optional: use it if present, otherwise boot without initrd
if fatload mmc 1 ${initrd_loadaddr} uInitrd; then
    setenv boot_start bootm ${kernel_loadaddr} ${initrd_loadaddr} ${dtb_loadaddr}
else
    setenv boot_start bootm ${kernel_loadaddr} - ${dtb_loadaddr}
fi

# Load kernel and dtb, then boot
if fatload mmc 1 ${kernel_loadaddr} uImage; then
    if fatload mmc 1 ${dtb_loadaddr} ${dtb_name}; then
        run boot_start
    else
        imgread dtb boot ${loadaddr} ${dtb_loadaddr}
        run boot_start
    fi
fi
